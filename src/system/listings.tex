% ============================================================
% Листинги исходного кода
% ============================================================

\RequirePackage{listings}
\RequirePackage{xcolor}
\RequirePackage{caption}
\RequirePackage{chngcntr}

% ------------------------------------------------------------
% Настройки нумерации
% ------------------------------------------------------------

\newif\ifConfigListingNumberedBySection
\ConfigListingNumberedBySectionfalse

% ------------------------------------------------------------
% Цвета
% ------------------------------------------------------------

\definecolor{codeBackground}{RGB}{248,248,248}
\definecolor{codeFrame}{RGB}{220,220,220}
\definecolor{codeKeyword}{RGB}{0,80,180}
\definecolor{codeString}{RGB}{0,120,70}
\definecolor{codeComment}{RGB}{100,100,100}
\definecolor{codeNumber}{RGB}{140,140,140}

% ------------------------------------------------------------
% Стиль листинга
% ------------------------------------------------------------

\lstset{
    backgroundcolor=\color{codeBackground},
    frame=single,
    rulecolor=\color{codeFrame},
    numbers=left,
    numberstyle=\scriptsize\color{codeNumber},
    numbersep=12pt,
    basicstyle=\ttfamily\small,
    keywordstyle=\color{codeKeyword},
    stringstyle=\color{codeString},
    commentstyle=\color{codeComment},
    breaklines=true,
    keepspaces=true,
    showstringspaces=false,
    tabsize=4,
    xleftmargin=2em,
    framexleftmargin=1.5em,
    captionpos=b
}

% ------------------------------------------------------------
% Нумерация листингов
% ------------------------------------------------------------

\newif\ifConfigListingNumberedBySection
\ConfigListingNumberedBySectionfalse

\AtBeginDocument{%
    \ifConfigListingNumberedBySection
    \counterwithin{lstlisting}{section}%
    \fi
}
% ------------------------------------------------------------
% Настройка подписей
% ------------------------------------------------------------

\newcommand{\ESKDconfigureListingCaption}{%
    \captionsetup[lstlisting]{
        labelsep=\ConfigListingCaptionLabelSep,
        justification=\ConfigListingCaptionJustification,
        singlelinecheck=\ConfigListingCaptionSingleLineCheck,
        font=\ConfigListingCaptionFont
    }
}
